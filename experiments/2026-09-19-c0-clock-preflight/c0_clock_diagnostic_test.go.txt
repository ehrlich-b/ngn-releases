const (
	c0RodentModelEnv = "RODENT_V11_ANAND_MODEL"
	c0RodentModelSHA = "5f7480b56538e9e64ee02fb5b4842bf0394ee44b707dae3a63ec7f902abd0afb"
	c0HashMB         = 128
	c0FixedNodes     = uint64(400_000)
)

var c0WarmFixtures = []struct {
	name string
	ply  int
	fen  string
}{
	{"start", 0, "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1"},
	{"accepted_pair_ply23", 23, "r2Qkb1r/p4ppp/2p1p3/8/8/2P3P1/P1P2P1P/R1B1K2R b KQkq - 0 1"},
	{"accepted_game1_ply48", 48, "5R2/p1kr3p/2p3p1/P1b1pp2/8/2P3PP/2P1KP2/2B5 w - - 1 1"},
	{"accepted_game2_ply48", 48, "5rk1/5pp1/R1p1p3/4b2p/P1K5/1RP1B1P1/r1P2P1P/8 w - - 4 1"},
}

func loadC0RodentModel(tb testing.TB) *rodenteval.Model {
	tb.Helper()
	path := os.Getenv(c0RodentModelEnv)
	if path == "" {
		tb.Fatalf("set %s", c0RodentModelEnv)
	}
	model, err := rodenteval.LoadV11Anand(path)
	if err != nil {
		tb.Fatal(err)
	}
	metadata, err := model.Metadata()
	if err != nil {
		tb.Fatal(err)
	}
	if metadata.SHA256 != c0RodentModelSHA || metadata.Bytes != 789568 {
		tb.Fatalf("Rodent V1.1 identity sha=%s bytes=%d", metadata.SHA256, metadata.Bytes)
	}
	return model
}

func newC0RodentSearcher(tb testing.TB, model *rodenteval.Model, maxNodes uint64) *SearchEngine {
	tb.Helper()
	searcher, err := NewSearchEngineWithHash(c0HashMB)
	if err != nil {
		tb.Fatal(err)
	}
	if err := searcher.SelectRodentV11AnandEvaluator(model); err != nil {
		tb.Fatal(err)
	}
	if err := searcher.ConfigureThreads(1); err != nil {
		tb.Fatal(err)
	}
	searcher.SetMaxNodes(maxNodes)
	return searcher
}

func requireC0Result(tb testing.TB, position *Position, before pvPositionSnapshot, result *SearchInfo) {
	tb.Helper()
	if result == nil || result.BestMove == EmptyMove || result.EffectiveThreads != 1 {
		tb.Fatalf("invalid search result: %+v", result)
	}
	legal := false
	for _, move := range GenerateLegalMoves(position) {
		if move == result.BestMove {
			legal = true
			break
		}
	}
	if !legal {
		tb.Fatalf("best move %v is not legal", result.BestMove)
	}
	if !reflect.DeepEqual(snapshotPVPosition(position), before) {
		tb.Fatal("search changed root position")
	}
}

func TestC0WarmTournamentStopMechanism(t *testing.T) {
	if runtime.GOMAXPROCS(0) != 1 {
		t.Fatalf("GOMAXPROCS=%d, want 1", runtime.GOMAXPROCS(0))
	}
	model := loadC0RodentModel(t)
	searcher := newC0RodentSearcher(t, model, 0)
	params := SearchParams{
		WhiteTime: 10_000,
		BlackTime: 10_000,
		WhiteInc:  100,
		BlackInc:  100,
	}

	const passes = 3
	for pass := 1; pass <= passes; pass++ {
		for _, fixture := range c0WarmFixtures {
			position, err := ParseFEN(fixture.fen)
			if err != nil {
				t.Fatal(err)
			}
			before := snapshotPVPosition(position)
			tm := NewTimeManager()
			tm.UpdateGameState(fixture.ply, 0, 0, position)
			var observer searchTerminationObserver
			tm.setSearchTerminationObserver(&observer)
			tm.SetTimeControl(params, position.Turn() == White)
			result := searcher.SearchIterativeDeepening(position, MaximumDepth, tm)
			requireC0Result(t, position, before, result)
			if observer.FirstReason == searchTerminationNone || observer.IterationOverflow != 0 {
				t.Fatalf("pass=%d fixture=%s incomplete observer: %+v", pass, fixture.name, observer)
			}
			lastCompleted := time.Duration(0)
			lastDuration := time.Duration(0)
			lastAttempts := 0
			if observer.IterationCount > 0 {
				last := observer.Iterations[observer.IterationCount-1]
				lastCompleted = last.CompletedAtElapsed
				lastDuration = last.Duration
				lastAttempts = last.AspirationAttempts
			}
			tail := observer.FirstElapsed - lastCompleted
			t.Logf("C0 pass=%d fixture=%s reason=%s stop_depth=%d elapsed_us=%d soft_us=%d hard_us=%d completed_depth=%d iterations=%d last_iteration_us=%d last_attempts=%d tail_us=%d nodes=%d qnodes=%d move=%v score=%d",
				pass, fixture.name, observer.FirstReason, observer.FirstDepth,
				observer.FirstElapsed.Microseconds(), tm.softTime.Microseconds(), tm.hardTime.Microseconds(),
				result.Depth, observer.IterationCount, lastDuration.Microseconds(), lastAttempts,
				tail.Microseconds(), result.Nodes, result.QNodes, result.BestMove, result.BestScore)
		}
	}
}

func BenchmarkC0FixedNodeObserver(b *testing.B) {
	if runtime.GOMAXPROCS(0) != 1 {
		b.Fatalf("GOMAXPROCS=%d, want 1", runtime.GOMAXPROCS(0))
	}
	model := loadC0RodentModel(b)
	for _, enabled := range []bool{false, true} {
		name := "off"
		if enabled {
			name = "on"
		}
		b.Run(name, func(b *testing.B) {
			b.ReportAllocs()
			for iteration := 0; iteration < b.N; iteration++ {
				b.StopTimer()
				searcher := newC0RodentSearcher(b, model, c0FixedNodes)
				position := newUCIStartingPosition()
				before := snapshotPVPosition(position)
				tm := NewTimeManager()
				tm.SetTimeControl(SearchParams{Depth: MaximumDepth}, true)
				var observer searchTerminationObserver
				if enabled {
					tm.setSearchTerminationObserver(&observer)
				}
				b.StartTimer()
				result := searcher.SearchIterativeDeepening(position, MaximumDepth, tm)
				b.StopTimer()
				requireC0Result(b, position, before, result)
				if !result.Stopped || result.Nodes < c0FixedNodes {
					b.Fatalf("incomplete fixed-node result: %+v", result)
				}
				if enabled && observer.FirstReason != searchTerminationNodeCap {
					b.Fatalf("observer reason=%s, want node-cap", observer.FirstReason)
				}
			}
		})
	}
}
