package main

import (
	"bufio"
	"crypto/sha256"
	"encoding/hex"
	"encoding/json"
	"fmt"
	"github.com/ehrlich-b/ngn/engine"
	"math"
	"math/rand"
	"os"
	"strconv"
	"strings"
)

type Example struct {
	FEN   string  `json:"fen"`
	Label float64 `json:"label"`
}
type Report struct {
	File                                                                                       string
	SHA256                                                                                     string
	LoaderError                                                                                string
	Rows, BadRows, OutOfRangeLabels, UniqueInputs, DuplicateRows, ConflictingDuplicateRows     int
	Labels                                                                                     map[string]int
	FullmoveOne, NoCastling, WhiteToMove                                                       int
	UniqueBoardsAfterInputDedup, RepeatedBoardsAfterInputDedup                                 int
	Train, Validation, ValidationWithTrainingBoard                                             int
	ReservoirSize, QuietAccepted, InCheck, PreviousMoverInCheck, ClockSensitive, BoardMutation int
	RejectedExamples                                                                           []Example
}

func main() {
	var reports []Report
	for _, path := range os.Args[1:] {
		content, err := os.ReadFile(path)
		if err != nil {
			panic(err)
		}
		h := sha256.Sum256(content)
		r := Report{File: path, SHA256: hex.EncodeToString(h[:]), Labels: map[string]int{}}
		seen := map[uint64]float64{}
		f, err := os.Open(path)
		if err != nil {
			panic(err)
		}
		scan := bufio.NewScanner(f)
		rng := rand.New(rand.NewSource(2800))
		reservoir := []Example{}
		for scan.Scan() {
			line := strings.TrimSpace(scan.Text())
			if line == "" || strings.HasPrefix(line, "#") {
				continue
			}
			fields := strings.Fields(line)
			r.Rows++
			if len(fields) != 7 {
				r.BadRows++
				continue
			}
			label, err := strconv.ParseFloat(fields[6], 64)
			if err != nil || math.IsNaN(label) || math.IsInf(label, 0) {
				r.BadRows++
				continue
			}
			if label < 0 || label > 1 {
				r.OutOfRangeLabels++
			}
			labelKey := fields[6]
			if label != 0 && label != 0.5 && label != 1 {
				labelKey = "soft"
			}
			r.Labels[labelKey]++
			if fields[5] == "1" {
				r.FullmoveOne++
			}
			if fields[2] == "-" {
				r.NoCastling++
			}
			if fields[1] == "w" {
				r.WhiteToMove++
			}
			fen := strings.Join(fields[:6], " ")
			pos, err := engine.ParseFEN(fen)
			if err != nil {
				r.BadRows++
				continue
			}
			if old, ok := seen[pos.Hash()]; ok {
				r.DuplicateRows++
				if old != label {
					r.ConflictingDuplicateRows++
				}
			} else {
				seen[pos.Hash()] = label
				r.UniqueInputs++
			}
			ex := Example{fen, label}
			if len(reservoir) < 2000 {
				reservoir = append(reservoir, ex)
			} else if i := rng.Intn(r.Rows); i < len(reservoir) {
				reservoir[i] = ex
			}
		}
		if err := scan.Err(); err != nil {
			panic(err)
		}
		f.Close()
		samples, ndup, err := engine.LoadTexelSamples(path)
		if err != nil {
			r.LoaderError = err.Error()
			reports = append(reports, r)
			fmt.Fprintln(os.Stderr, "loader rejected", path, err)
			continue
		}
		if ndup != r.DuplicateRows || len(samples) != r.UniqueInputs {
			panic("audit/loader count mismatch")
		}
		boards := map[engine.Bitboard]bool{}
		for _, s := range samples {
			if boards[s.Board] {
				r.RepeatedBoardsAfterInputDedup++
			}
			boards[s.Board] = true
		}
		r.UniqueBoardsAfterInputDedup = len(boards)
		train, test := engine.SplitTrainTest(samples, .2, 2800)
		r.Train = len(train)
		r.Validation = len(test)
		trainingBoards := map[engine.Bitboard]bool{}
		for _, s := range train {
			trainingBoards[s.Board] = true
		}
		for _, s := range test {
			if trainingBoards[s.Board] {
				r.ValidationWithTrainingBoard++
			}
		}
		for _, ex := range reservoir {
			pos, _ := engine.ParseFEN(ex.FEN)
			before := pos.Copy()
			r.ReservoirSize++
			if pos.IsInCheck() {
				r.InCheck++
			}
			fields := strings.Fields(ex.FEN)
			if fields[1] == "w" {
				fields[1] = "b"
			} else {
				fields[1] = "w"
			}
			other, _ := engine.ParseFEN(strings.Join(fields, " "))
			if other.IsInCheck() {
				r.PreviousMoverInCheck++
			}
			quiet := engine.IsQuietPosition(pos)
			if quiet {
				r.QuietAccepted++
			} else if len(r.RejectedExamples) < 5 {
				r.RejectedExamples = append(r.RejectedExamples, ex)
			}
			if !engine.PositionEqual(before, pos) {
				r.BoardMutation++
			}
			pos.HalfMoveClock = 0
			if engine.IsQuietPosition(pos) != quiet {
				r.ClockSensitive++
			}
		}
		reports = append(reports, r)
		fmt.Fprintf(os.Stderr, "audited %s: %d rows; %d/%d held-out boards in training; quiet %d/%d\n", path, r.Rows, r.ValidationWithTrainingBoard, r.Validation, r.QuietAccepted, r.ReservoirSize)
	}
	enc := json.NewEncoder(os.Stdout)
	enc.SetIndent("", "  ")
	if err := enc.Encode(reports); err != nil {
		panic(err)
	}
}
