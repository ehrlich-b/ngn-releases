const (
	counterProfilePrefixesEnv = "NGN_COUNTER_PROFILE_PREFIXES"
	counterProfileOutputEnv   = "NGN_COUNTER_PROFILE_OUTPUT"
	counterProfileStartFEN    = "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1"
)

type counterProfilePrefixCorpus struct {
	Schema          string                       `json:"schema"`
	SourcePGN       string                       `json:"source_pgn"`
	SourcePGNSHA256 string                       `json:"source_pgn_sha256"`
	Selection       string                       `json:"selection"`
	TargetAfterPly  []int                        `json:"target_after_plies"`
	Records         []counterProfilePrefixRecord `json:"records"`
}

type counterProfilePrefixRecord struct {
	GameIndex              int      `json:"game_index"`
	Round                  string   `json:"round"`
	White                  string   `json:"white"`
	Black                  string   `json:"black"`
	ResultRecordedNotChosen string   `json:"result_recorded_not_selected"`
	FullGamePlyCount       int      `json:"full_game_ply_count"`
	MovesThroughPly48      []string `json:"moves_through_ply_48"`
	MovesSHA256            string   `json:"moves_sha256"`
}

type counterProfilePersistentRoot struct {
	GameIndex    int                                `json:"game_index"`
	AfterPly     int                                `json:"after_ply"`
	FEN          string                             `json:"fen"`
	PositionHash string                             `json:"position_hash"`
	MovePrefix   []string                           `json:"move_prefix"`
	Trajectory   counterTransitionProfileTrajectory `json:"trajectory"`
}

type counterProfilePersistentReceipt struct {
	Schema         string                         `json:"schema"`
	ModelSHA256    string                         `json:"model_sha256"`
	HashMB         int                            `json:"hash_mb"`
	Threads        int                            `json:"threads"`
	NodeLimit      uint64                         `json:"node_limit"`
	PrefixCorpus   counterProfilePrefixCorpus     `json:"prefix_corpus"`
	PersistentRoot []counterProfilePersistentRoot `json:"persistent_roots"`
}

func loadCounterProfilePrefixes(tb testing.TB) counterProfilePrefixCorpus {
	tb.Helper()
	path := os.Getenv(counterProfilePrefixesEnv)
	if path == "" {
		tb.Fatalf("set %s to the frozen accepted-game prefix fixture", counterProfilePrefixesEnv)
	}
	encoded, err := os.ReadFile(path)
	if err != nil {
		tb.Fatal(err)
	}
	var corpus counterProfilePrefixCorpus
	if err := json.Unmarshal(encoded, &corpus); err != nil {
		tb.Fatal(err)
	}
	if corpus.Schema != "ngn-counter-profile-prefixes-v1" ||
		corpus.SourcePGNSHA256 != "35bdd00b28f585224e6c04392bdbf79cd4472c9ea7549b84acfc4379b07847d2" ||
		corpus.Selection != "first three games in immutable PGN; no result/position selection" ||
		len(corpus.TargetAfterPly) != 2 || corpus.TargetAfterPly[0] != 23 || corpus.TargetAfterPly[1] != 48 ||
		len(corpus.Records) != 3 {
		tb.Fatalf("wrong frozen prefix corpus identity: %+v", corpus)
	}
	for index, record := range corpus.Records {
		digest := sha256.Sum256([]byte(joinMoves(record.MovesThroughPly48) + "\n"))
		if record.GameIndex != index+1 || record.FullGamePlyCount < 48 || len(record.MovesThroughPly48) != 48 ||
			hex.EncodeToString(digest[:]) != record.MovesSHA256 {
			tb.Fatalf("wrong game %d prefix identity", index+1)
		}
	}
	return corpus
}

func joinMoves(moves []string) string {
	if len(moves) == 0 {
		return ""
	}
	joined := moves[0]
	for _, move := range moves[1:] {
		joined += " " + move
	}
	return joined
}

func newCounterProfileGame(tb testing.TB, model *countereval.Model) (*SearchEngine, *Position) {
	tb.Helper()
	searcher, position, _ := newCounterTransitionProfileSearch(tb, model, counterProfileStartFEN)
	searcher.NewGame()
	return searcher, position
}

func replayCounterProfileMoves(tb testing.TB, searcher *SearchEngine, position *Position, moves []string) {
	tb.Helper()
	for index, text := range moves {
		move, err := ParseUCIMove(position, text)
		if err != nil {
			tb.Fatalf("replay ply %d %s: %v", index+1, text, err)
		}
		if _, _, _, ok := position.GameMakeMove(move); !ok {
			tb.Fatalf("replay ply %d %s was rejected", index+1, text)
		}
		searcher.SetLastMovePlayed(move)
	}
}

func prepareCounterProfileRoot(tb testing.TB, model *countereval.Model, position *Position) (pvPositionSnapshot, countereval.Board, float32) {
	tb.Helper()
	before := snapshotPVPosition(position)
	board, err := counterBoardFromPosition(position)
	if err != nil {
		tb.Fatal(err)
	}
	wantRaw, err := model.EvaluateFullRefresh(board)
	if err != nil {
		tb.Fatal(err)
	}
	return before, board, wantRaw
}

func runLabelledCounterProfileSearch(searcher *SearchEngine, position *Position, family, fixture string) *SearchInfo {
	var final *SearchInfo
	labels := pprof.Labels("scope", "search", "family", family, "fixture", fixture)
	pprof.Do(context.Background(), labels, func(context.Context) {
		final = searcher.SearchIterativeDeepening(position, MaximumDepth, nil)
	})
	return final
}

func requireCounterProfileSearch(tb testing.TB, searcher *SearchEngine, position *Position, before pvPositionSnapshot, board countereval.Board, wantRaw float32, final *SearchInfo) {
	tb.Helper()
	if final == nil || !final.Stopped || final.Nodes < counterTransitionProfileNodes || final.EffectiveThreads != 1 {
		tb.Fatalf("incomplete profile search: %+v", final)
	}
	// Nil-callback profiling deliberately avoids reporting-only TT/PV refresh.
	// The separate snapshot test exercises and validates every full PV.
	legalBestMove := false
	for _, move := range GenerateLegalMoves(position) {
		if move == final.BestMove {
			legalBestMove = true
			break
		}
	}
	if final.BestMove == EmptyMove || !legalBestMove {
		tb.Fatalf("profile search returned illegal best move %s", final.BestMove.ToString())
	}
	if !reflect.DeepEqual(snapshotPVPosition(position), before) {
		tb.Fatal("profile search changed root position/history")
	}
	context := searcher.worker.evaluator.counterContext
	if context == nil || context.Depth() != 0 || context.Board() != board ||
		math.Float32bits(context.EvaluateRaw()) != math.Float32bits(wantRaw) {
		tb.Fatal("profile search did not restore exact Counter root/context")
	}
}

func TestCounterProfilePersistentSnapshot(t *testing.T) {
	model, metadata := loadCounterTransitionProfileModel(t)
	output := os.Getenv(counterProfileOutputEnv)
	if output == "" {
		t.Fatalf("set %s to a new explicit receipt path", counterProfileOutputEnv)
	}
	corpus := loadCounterProfilePrefixes(t)
	receipt := counterProfilePersistentReceipt{
		Schema: "ngn-counter-profile-persistent-snapshot-v1", ModelSHA256: metadata.SHA256,
		HashMB: counterTransitionProfileHashMB, Threads: 1, NodeLimit: counterTransitionProfileNodes,
		PrefixCorpus: corpus,
	}
	for _, game := range corpus.Records {
		searcher, position := newCounterProfileGame(t, model)
		previous := 0
		for _, afterPly := range corpus.TargetAfterPly {
			replayCounterProfileMoves(t, searcher, position, game.MovesThroughPly48[previous:afterPly])
			before, board, wantRaw := prepareCounterProfileRoot(t, model, position)
			trajectory := runCounterTransitionProfileTrajectory(t, searcher, position, before, board, wantRaw, true)
			receipt.PersistentRoot = append(receipt.PersistentRoot, counterProfilePersistentRoot{
				GameIndex: game.GameIndex, AfterPly: afterPly, FEN: GenerateFEN(position),
				PositionHash: fmt.Sprintf("%016x", position.Hash()),
				MovePrefix: append([]string(nil), game.MovesThroughPly48[:afterPly]...), Trajectory: trajectory,
			})
			previous = afterPly
		}
	}
	encoded, err := json.MarshalIndent(receipt, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	encoded = append(encoded, '\n')
	file, err := os.OpenFile(output, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0o644)
	if err != nil {
		t.Fatal(err)
	}
	if _, err := file.Write(encoded); err != nil {
		_ = file.Close()
		t.Fatal(err)
	}
	if err := file.Close(); err != nil {
		t.Fatal(err)
	}
}

func BenchmarkCounterProfileCold(b *testing.B) {
	model, _ := loadCounterTransitionProfileModel(b)
	for _, fixture := range counterTransitionProfileFixtures {
		fixture := fixture
		b.Run(fixture.name, func(b *testing.B) {
			b.ReportAllocs()
			for iteration := 0; iteration < b.N; iteration++ {
				b.StopTimer()
				searcher, position, _ := newCounterTransitionProfileSearch(b, model, fixture.fen)
				before, board, wantRaw := prepareCounterProfileRoot(b, model, position)
				b.StartTimer()
				final := runLabelledCounterProfileSearch(searcher, position, "cold", fixture.name)
				b.StopTimer()
				requireCounterProfileSearch(b, searcher, position, before, board, wantRaw, final)
			}
			b.ReportMetric(float64(counterTransitionProfileNodes), "minimum_nodes/op")
		})
	}
}

func BenchmarkCounterProfilePersistent(b *testing.B) {
	model, _ := loadCounterTransitionProfileModel(b)
	corpus := loadCounterProfilePrefixes(b)
	for _, game := range corpus.Records {
		game := game
		b.Run(fmt.Sprintf("game_%d", game.GameIndex), func(b *testing.B) {
			b.ReportAllocs()
			for iteration := 0; iteration < b.N; iteration++ {
				b.StopTimer()
				searcher, position := newCounterProfileGame(b, model)
				replayCounterProfileMoves(b, searcher, position, game.MovesThroughPly48[:23])
				before, board, wantRaw := prepareCounterProfileRoot(b, model, position)
				b.StartTimer()
				first := runLabelledCounterProfileSearch(searcher, position, "persistent", fmt.Sprintf("game_%d_ply_23", game.GameIndex))
				b.StopTimer()
				requireCounterProfileSearch(b, searcher, position, before, board, wantRaw, first)

				replayCounterProfileMoves(b, searcher, position, game.MovesThroughPly48[23:48])
				before, board, wantRaw = prepareCounterProfileRoot(b, model, position)
				b.StartTimer()
				second := runLabelledCounterProfileSearch(searcher, position, "persistent", fmt.Sprintf("game_%d_ply_48", game.GameIndex))
				b.StopTimer()
				requireCounterProfileSearch(b, searcher, position, before, board, wantRaw, second)
			}
			b.ReportMetric(2*float64(counterTransitionProfileNodes), "minimum_nodes/op")
			b.ReportMetric(2, "searches/op")
		})
	}
}
