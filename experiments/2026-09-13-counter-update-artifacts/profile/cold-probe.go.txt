const (
	counterTransitionProfileModelEnv  = "NGN_COUNTER55_PROFILE_MODEL"
	counterTransitionProfileOutputEnv = "NGN_COUNTER55_PROFILE_OUTPUT"
	counterTransitionProfileModelSHA  = "3488baed71f4d432d028e05b65e2ff13f8c9afe7f91974d1a3c34a0691d6670c"
	counterTransitionProfileNodes     = uint64(400_000)
	counterTransitionProfileHashMB    = 128
)

	{"kiwipete_black", "r3k2r/p1ppqpb1/bn2pnp1/2Pp4/1p2P3/2N2N2/PPQBBPPP/R3K2R b KQkq - 37 1"},

type counterTransitionProfileInfo struct {
	Scalars map[string]any `json:"scalars"`
	PV      []string       `json:"pv"`
}

type counterTransitionProfileTrajectory struct {
	Callbacks     []counterTransitionProfileInfo `json:"callbacks"`
	Final         counterTransitionProfileInfo   `json:"final"`
	RootRawBits   uint32                         `json:"root_raw_bits"`
	RootRestored  bool                           `json:"root_restored"`
	ContextDepth  int                            `json:"context_depth"`
	ContextAtRoot bool                           `json:"context_at_root"`
}

type counterTransitionProfileFixtureResult struct {
	Name string                             `json:"name"`
	FEN  string                             `json:"fen"`
	Cold counterTransitionProfileTrajectory `json:"cold"`
	Warm counterTransitionProfileTrajectory `json:"warm"`
}

type counterTransitionProfileReceipt struct {
	Schema      int                                     `json:"schema"`
	ModelSHA256 string                                  `json:"model_sha256"`
	ModelBytes  int                                     `json:"model_bytes"`
	HashMB      int                                     `json:"hash_mb"`
	Threads     int                                     `json:"threads"`
	GOMAXPROCS  int                                     `json:"gomaxprocs"`
	NodeLimit   uint64                                  `json:"node_limit"`
	Fixtures    []counterTransitionProfileFixtureResult `json:"fixtures"`
}

func loadCounterTransitionProfileModel(tb testing.TB) (*countereval.Model, countereval.LoadMetadata) {
	tb.Helper()
	path := os.Getenv(counterTransitionProfileModelEnv)
	if path == "" {
		tb.Skipf("set %s to the accepted Counter 5.5 model", counterTransitionProfileModelEnv)
	}
	file, err := os.Open(path)
	if err != nil {
		tb.Fatal(err)
	}
	defer file.Close()
	model, metadata, err := countereval.LoadCounter55Legacy(file)
	if err != nil {
		tb.Fatal(err)
	}
	if metadata.SHA256 != counterTransitionProfileModelSHA || metadata.Bytes != countereval.LegacyFileSize {
		tb.Fatalf("Counter model identity sha=%s bytes=%d", metadata.SHA256, metadata.Bytes)
	}
	return model, metadata
}

func newCounterTransitionProfileSearch(tb testing.TB, model *countereval.Model, fen string) (*SearchEngine, *Position, pvPositionSnapshot) {
	tb.Helper()
	if got := runtime.GOMAXPROCS(0); got != 1 {
		tb.Fatalf("GOMAXPROCS=%d, want 1", got)
	}
	searcher, err := NewSearchEngineWithHash(counterTransitionProfileHashMB)
	if err != nil {
		tb.Fatal(err)
	}
	if err := searcher.SelectCounter55Evaluator(model); err != nil {
		tb.Fatal(err)
	}
	if err := searcher.ConfigureThreads(1); err != nil {
		tb.Fatal(err)
	}
	searcher.SetMaxNodes(counterTransitionProfileNodes)
	pos, err := ParseFEN(fen)
	if err != nil {
		tb.Fatal(err)
	}
	return searcher, pos, snapshotPVPosition(pos)
}

func counterTransitionProfileSnapshot(info *SearchInfo) counterTransitionProfileInfo {
	scalars := make(map[string]any)
	value := reflect.ValueOf(info).Elem()
	typeInfo := value.Type()
	for index := 0; index < value.NumField(); index++ {
		fieldType := typeInfo.Field(index)
		if fieldType.PkgPath != "" {
			continue
		}
		field := value.Field(index)
		switch field.Kind() {
		case reflect.Bool, reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64,
			reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64:
			scalars[fieldType.Name] = field.Interface()
		case reflect.Array:
			switch fieldType.Name {
			case "CutIdxHist", "BcutBand", "FmcBand", "CutMissByClass":
				scalars[fieldType.Name] = field.Interface()
			}
		}
	}
	pv := make([]string, info.PVLength)
	for index := range pv {
		pv[index] = MoveToAlgebraic(info.PV[index])
	}
	return counterTransitionProfileInfo{Scalars: scalars, PV: pv}
}

func requireCounterTransitionProfilePV(tb testing.TB, root *Position, info *SearchInfo) {
	tb.Helper()
	if info.PVLength <= 0 || info.PVLength > MaximumDepth {
		tb.Fatalf("PV length=%d", info.PVLength)
	}
	if info.BestMove == EmptyMove || info.PV[0] != info.BestMove {
		tb.Fatalf("best move=%v PV[0]=%v", info.BestMove, info.PV[0])
	}
	pos := root.Copy()
	for ply := 0; ply < info.PVLength; ply++ {
		move := info.PV[ply]
		var generated [256]Move
		count := GenerateMovesIntoBuffer(pos, generated[:])
		found := false
		for _, candidate := range generated[:count] {
			if candidate == move {
				found = true
				break
			}
		}
		if !found || (move.IsCastle() && !isLegalCastle(pos, move)) {
			tb.Fatalf("illegal PV move at ply %d: %s", ply, MoveToAlgebraic(move))
		}
		mover := move.MovingPiece().Color()
		_, _, _, made := pos.MakeMove(move)
		if !made || isInCheck(pos, mover) {
			tb.Fatalf("PV move failed at ply %d: %s", ply, MoveToAlgebraic(move))
		}
	}
}

func runCounterTransitionProfileTrajectory(tb testing.TB, searcher *SearchEngine, pos *Position, before pvPositionSnapshot, rootBoard countereval.Board, wantRaw float32, capture bool) counterTransitionProfileTrajectory {
	tb.Helper()
	var callbacks []counterTransitionProfileInfo
	var callback DepthCallback
	if capture {
		callback = func(info *SearchInfo) {
			requireCounterTransitionProfilePV(tb, pos, info)
			callbacks = append(callbacks, counterTransitionProfileSnapshot(info))
		}
	}
	final := searcher.SearchIterativeDeepeningWithCallback(pos, MaximumDepth, nil, callback)
	if final == nil {
		tb.Fatal("nil search result")
	}
	if !final.Stopped || final.Nodes < counterTransitionProfileNodes || final.EffectiveThreads != 1 {
		tb.Fatalf("stopped=%v nodes=%d effective=%d", final.Stopped, final.Nodes, final.EffectiveThreads)
	}
	requireCounterTransitionProfilePV(tb, pos, final)
	restored := reflect.DeepEqual(snapshotPVPosition(pos), before)
	if !restored {
		tb.Fatal("search changed root position")
	}
	if searcher.worker.evaluator == nil || searcher.worker.evaluator.counterContext == nil {
		tb.Fatal("Counter search has no Counter context")
	}
	context := searcher.worker.evaluator.counterContext
	depth := context.Depth()
	if depth != 0 {
		tb.Fatalf("Counter context depth=%d", depth)
	}
	boardAtRoot := context.Board() == rootBoard
	if !boardAtRoot {
		tb.Fatal("Counter context board differs from exact root board")
	}
	gotRaw := context.EvaluateRaw()
	if math.Float32bits(gotRaw) != math.Float32bits(wantRaw) {
		tb.Fatalf("Counter root raw bits=%08x, want %08x", math.Float32bits(gotRaw), math.Float32bits(wantRaw))
	}
	result := counterTransitionProfileTrajectory{
		RootRawBits: math.Float32bits(gotRaw), RootRestored: restored,
		ContextDepth: depth, ContextAtRoot: boardAtRoot,
	}
	if capture {
		if len(callbacks) == 0 {
			tb.Fatal("no completed-depth callbacks")
		}
		result.Callbacks = callbacks
		result.Final = counterTransitionProfileSnapshot(final)
	}
	return result
}

func runCounterTransitionProfileFixture(tb testing.TB, model *countereval.Model, name, fen string) counterTransitionProfileFixtureResult {
	tb.Helper()
	searcher, pos, before := newCounterTransitionProfileSearch(tb, model, fen)
	rootBoard, err := counterBoardFromPosition(pos)
	if err != nil {
		tb.Fatal(err)
	}
	wantRaw, err := model.EvaluateFullRefresh(rootBoard)
	if err != nil {
		tb.Fatal(err)
	}
	cold := runCounterTransitionProfileTrajectory(tb, searcher, pos, before, rootBoard, wantRaw, true)
	warm := runCounterTransitionProfileTrajectory(tb, searcher, pos, before, rootBoard, wantRaw, true)
	return counterTransitionProfileFixtureResult{Name: name, FEN: fen, Cold: cold, Warm: warm}
}

// TestCounterTransitionFixedNodeSnapshot is compiled unchanged into parent and
// candidate test binaries. Byte equality of its JSON output is the correctness
// gate before either binary contributes wall-time or profile evidence.
func TestCounterTransitionFixedNodeSnapshot(t *testing.T) {
	model, metadata := loadCounterTransitionProfileModel(t)
	output := os.Getenv(counterTransitionProfileOutputEnv)
	if output == "" {
		t.Fatalf("set %s to a new explicit receipt path", counterTransitionProfileOutputEnv)
	}
	receipt := counterTransitionProfileReceipt{
		Schema: 1, ModelSHA256: metadata.SHA256, ModelBytes: metadata.Bytes,
		HashMB: counterTransitionProfileHashMB, Threads: 1, GOMAXPROCS: runtime.GOMAXPROCS(0), NodeLimit: counterTransitionProfileNodes,
	}
	for _, fixture := range counterTransitionProfileFixtures {
		receipt.Fixtures = append(receipt.Fixtures,
			runCounterTransitionProfileFixture(t, model, fixture.name, fixture.fen))
	}
	data, err := json.MarshalIndent(receipt, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	data = append(data, '\n')
	file, err := os.OpenFile(output, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0o644)
	if err != nil {
		t.Fatal(err)
	}
	if _, err := file.Write(data); err != nil {
		_ = file.Close()
		t.Fatal(err)
	}
	if err := file.Close(); err != nil {
		t.Fatal(err)
	}
}

// BenchmarkCounterTransitionFixedNodes uses the same source and fixture order
// for parent and candidate. Setup and invariant checks are excluded from the Go
// benchmark timer; CPU profiles remain separate processes from wall-time runs.
func BenchmarkCounterTransitionFixedNodes(b *testing.B) {
	model, _ := loadCounterTransitionProfileModel(b)
	for _, fixture := range counterTransitionProfileFixtures {
		fixture := fixture
		b.Run(fixture.name, func(b *testing.B) {
			b.ReportAllocs()
			for iteration := 0; iteration < b.N; iteration++ {
				b.StopTimer()
				searcher, pos, before := newCounterTransitionProfileSearch(b, model, fixture.fen)
				rootBoard, err := counterBoardFromPosition(pos)
				if err != nil {
					b.Fatal(err)
				}
				b.StartTimer()
				final := searcher.SearchIterativeDeepening(pos, MaximumDepth, nil)
				b.StopTimer()
				if final == nil || !final.Stopped || final.Nodes < counterTransitionProfileNodes {
					b.Fatalf("%s incomplete result %+v", fixture.name, final)
				}
				context := searcher.worker.evaluator.counterContext
				if !reflect.DeepEqual(snapshotPVPosition(pos), before) || context.Depth() != 0 || context.Board() != rootBoard {
					b.Fatalf("%s did not restore root/context", fixture.name)
				}
			}
			b.ReportMetric(float64(counterTransitionProfileNodes), "minimum_nodes/op")
		})
	}
}
