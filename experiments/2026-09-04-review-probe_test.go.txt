var reviewScoutSkips int
var reviewScoutFirst string

func TestReviewTraceParity(t *testing.T) {
	old := pawnCacheDisabled
	pawnCacheDisabled = true
	defer func() { pawnCacheDisabled = old }()
	rng := rand.New(rand.NewSource(2600))
	checked, bad, unexplained, maxDiff := 0, 0, 0, 0
	for game := 0; game < 16; game++ {
		p, _ := ParseFEN("rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1")
		for ply := 0; ply < 100; ply++ {
			tr := buildEvalTrace(&p.Board)
			want := evaluateUnsafe(&p.Board)
			got := reconstructEvalInt(&tr)
			checked++
			if got != want {
				bad++
				w, b := backwardPawnCounts(p.Board.GetBitboardOf(WhitePawn), p.Board.GetBitboardOf(BlackPawn))
				if got-want != (w-b)*taperW(backwardPawnPenalty, backwardPawnPenaltyEG, tr.Phase) {
					unexplained++
				}
				if abs(got-want) > maxDiff {
					maxDiff = abs(got - want)
				}
				if bad <= 3 {
					t.Logf("trace=%d live=%d diff=%d backward=%d/%d phase=%d FEN=%s", got, want, got-want, w, b, tr.Phase, GenerateFEN(p))
				}
			}
			moves := GenerateLegalMoves(p)
			if len(moves) == 0 {
				break
			}
			p.GameMakeMove(moves[rng.Intn(len(moves))])
		}
	}
	t.Logf("trace mismatch positions=%d/%d unexplained=%d maxAbsCp=%d", bad, checked, unexplained, maxDiff)
	if bad != 0 {
		t.Errorf("gradient trace does not reproduce the playing evaluator")
	}
}
func TestReviewScoutExtension(t *testing.T) {
	ClearStop()
	TranspositionTable = NewCache(DEFAULT_CACHE_SIZE)
	p, _ := ParseFEN("r3k2r/p1ppqpb1/bn2pnp1/3PN3/1p2P3/2N2Q1p/PPPBBPPP/R3K2R w KQkq - 0 1")
	info := SearchFixed(p, 6, nil)
	t.Logf("unextended non-first scouts=%d nodes=%d first-child=%s", reviewScoutSkips, info.Nodes, reviewScoutFirst)
	if reviewScoutSkips == 0 {
		t.Fatal("probe did not reach an extension omission")
	}
}

func TestReviewQuietFiltering(t *testing.T) {
	cases := []struct {
		fen  string
		want bool
	}{
		{"4k3/8/8/8/8/8/4P3/4K3 w - - 0 1", true},
		{"4k3/8/8/8/8/8/4P3/4K3 w - - 20 1", true},
		{"7k/8/8/8/8/8/p7/6K1 b - - 0 1", false},
	}
	for _, c := range cases {
		ClearStop()
		ClearHistoryTable()
		TranspositionTable = NewCache(DEFAULT_CACHE_SIZE)
		pos, _ := ParseFEN(c.fen)
		var caps [64]Move
		capN := GenerateCapturesIntoBuffer(pos, caps[:])
		static := EvaluateForPlayer(&pos.Board, pos.Turn())
		info := SearchInfo{CaptureBuffer: &caps}
		q := quiescence(pos, -INFINITY, INFINITY, 0, &info)
		TranspositionTable = NewCache(DEFAULT_CACHE_SIZE)
		got := IsQuietPosition(pos)
		t.Logf("quiet=%v want=%v static=%d q=%d captures=%d FEN=%s", got, c.want, static, q, capN, c.fen)
		if got != c.want {
			t.Errorf("quiet-filter classification mismatch")
		}
	}
}

	}
}
