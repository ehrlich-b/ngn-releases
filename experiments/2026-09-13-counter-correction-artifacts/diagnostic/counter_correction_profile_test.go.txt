const counterCorrectionProfileOutputEnv = "NGN_COUNTER_CORRECTION_PROFILE_OUTPUT"

type counterCorrectionProfileRoot struct {
	GameIndex          int                                `json:"game_index"`
	AfterPly           int                                `json:"after_ply"`
	FEN                string                             `json:"fen"`
	PositionHash       string                             `json:"position_hash"`
	Identity           bool                               `json:"off_on_identity"`
	ControlSHA256      string                             `json:"control_trajectory_sha256"`
	InstrumentedSHA256 string                             `json:"instrumented_trajectory_sha256"`
	CompletedDecisions counterTransitionProfileTrajectory `json:"completed_iteration_decisions"`
	Trace              correctionProfileStats             `json:"trace"`
}

type counterCorrectionProfileReceipt struct {
	Schema              string                         `json:"schema"`
	BaseCommit          string                         `json:"base_commit"`
	ModelSHA256         string                         `json:"model_sha256"`
	PrefixCorpus        counterProfilePrefixCorpus     `json:"prefix_corpus"`
	GOMAXPROCS          int                            `json:"gomaxprocs"`
	Threads             int                            `json:"threads"`
	HashMB              int                            `json:"hash_mb"`
	MinimumNodes        uint64                         `json:"minimum_nodes_per_search"`
	CallbackMode        string                         `json:"callback_mode"`
	CounterfactualScope string                         `json:"counterfactual_scope"`
	Roots               []counterCorrectionProfileRoot `json:"roots"`
}

func correctionProfileTrajectorySHA(tb testing.TB, trajectory counterTransitionProfileTrajectory) string {
	tb.Helper()
	encoded, err := json.Marshal(trajectory)
	if err != nil {
		tb.Fatal(err)
	}
	digest := sha256.Sum256(encoded)
	return hex.EncodeToString(digest[:])
}

func TestCounterCorrectionHistoryProfile(t *testing.T) {
	if runtime.GOMAXPROCS(0) != 1 {
		t.Fatalf("GOMAXPROCS=%d, want 1", runtime.GOMAXPROCS(0))
	}
	output := os.Getenv(counterCorrectionProfileOutputEnv)
	if output == "" {
		t.Fatalf("set %s to a new explicit receipt path", counterCorrectionProfileOutputEnv)
	}
	model, metadata := loadCounterTransitionProfileModel(t)
	corpus := loadCounterProfilePrefixes(t)
	receipt := counterCorrectionProfileReceipt{
		Schema:      "ngn-counter-correction-profile-v1",
		BaseCommit:  "270c73563139996d03f3c34b2840015b8e130df9",
		ModelSHA256: metadata.SHA256, PrefixCorpus: corpus,
		GOMAXPROCS: 1, Threads: 1, HashMB: counterTransitionProfileHashMB,
		MinimumNodes:        counterTransitionProfileNodes,
		CallbackMode:        "enabled identically in control and instrumented searches; reporting-only PV refresh included in both",
		CounterfactualScope: "local partial predicate opportunities on the baseline-reached path, holding actual improving and eligibility fixed; not full minor-off behavior, an ablation, or Elo evidence",
	}
	for _, game := range corpus.Records {
		control, controlPosition := newCounterProfileGame(t, model)
		instrumented, instrumentedPosition := newCounterProfileGame(t, model)
		previous := 0
		for _, afterPly := range corpus.TargetAfterPly {
			moves := game.MovesThroughPly48[previous:afterPly]
			replayCounterProfileMoves(t, control, controlPosition, moves)
			replayCounterProfileMoves(t, instrumented, instrumentedPosition, moves)
			controlBefore, controlBoard, controlRaw := prepareCounterProfileRoot(t, model, controlPosition)
			instrumentedBefore, instrumentedBoard, instrumentedRaw := prepareCounterProfileRoot(t, model, instrumentedPosition)
			if !reflect.DeepEqual(controlBefore, instrumentedBefore) || controlBoard != instrumentedBoard || controlRaw != instrumentedRaw {
				t.Fatalf("game %d ply %d: roots differ before search", game.GameIndex, afterPly)
			}
			controlTrajectory := runCounterTransitionProfileTrajectory(t, control, controlPosition,
				controlBefore, controlBoard, controlRaw, true)
			recorder := &correctionProfileRecorder{}
			instrumented.worker.history.correctionProfile = recorder
			instrumentedTrajectory := runCounterTransitionProfileTrajectory(t, instrumented, instrumentedPosition,
				instrumentedBefore, instrumentedBoard, instrumentedRaw, true)
			instrumented.worker.history.correctionProfile = nil
			controlSHA := correctionProfileTrajectorySHA(t, controlTrajectory)
			instrumentedSHA := correctionProfileTrajectorySHA(t, instrumentedTrajectory)
			if controlSHA != instrumentedSHA || !reflect.DeepEqual(controlTrajectory, instrumentedTrajectory) {
				t.Fatalf("game %d ply %d: instrumentation changed search: control=%s instrumented=%s",
					game.GameIndex, afterPly, controlSHA, instrumentedSHA)
			}
			if recorder.Stats.MainNodes == 0 || recorder.Stats.QNodes == 0 {
				t.Fatalf("game %d ply %d: recorder inactive: main=%d q=%d",
					game.GameIndex, afterPly, recorder.Stats.MainNodes, recorder.Stats.QNodes)
			}
			receipt.Roots = append(receipt.Roots, counterCorrectionProfileRoot{
				GameIndex: game.GameIndex, AfterPly: afterPly,
				FEN: GenerateFEN(controlPosition), PositionHash: fmt.Sprintf("%016x", controlPosition.Hash()),
				Identity: true, ControlSHA256: controlSHA, InstrumentedSHA256: instrumentedSHA,
				CompletedDecisions: instrumentedTrajectory, Trace: recorder.Stats,
			})
			previous = afterPly
		}
	}
	if len(receipt.Roots) != 6 {
		t.Fatalf("roots=%d, want 6", len(receipt.Roots))
	}
	encoded, err := json.MarshalIndent(receipt, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	encoded = append(encoded, '\n')
	file, err := os.OpenFile(output, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0o644)
	if err != nil {
		t.Fatal(err)
	}
	if _, err := file.Write(encoded); err != nil {
		_ = file.Close()
		t.Fatal(err)
	}
	if err := file.Close(); err != nil {
		t.Fatal(err)
	}
}
