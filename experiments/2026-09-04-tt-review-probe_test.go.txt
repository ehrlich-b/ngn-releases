package engine

import "testing"

func TestReviewQBudgetTT(t *testing.T) {
	const fen = "r3k2r/p1ppqpb1/bn2pnp1/3PN3/1p2P3/2N2Q1p/PPPBBPPP/R3K2R w KQkq - 0 1"
	p, err := ParseFEN(fen)
	if err != nil {
		t.Fatal(err)
	}
	before := p.Copy()
	ClearStop()
	ClearHistoryTable()
	TranspositionTable = NewCache(DEFAULT_CACHE_SIZE)
	shortInfo := newQSearchInfo()
	short := quiescenceWithDepth(p, -INFINITY, 100, 5, shortInfo, 5)
	if !PositionEqual(before, p) {
		t.Fatal("short search state changed")
	}
	move, score, depth, bound, hit, _ := TranspositionTable.Get(p.Hash())
	warmInfo := newQSearchInfo()
	warm := quiescenceWithDepth(p, -INFINITY, 100, 5, warmInfo, 0)
	if !PositionEqual(before, p) {
		t.Fatal("warm search state changed")
	}
	ClearStop()
	ClearHistoryTable()
	TranspositionTable = NewCache(DEFAULT_CACHE_SIZE)
	coldInfo := newQSearchInfo()
	cold := quiescenceWithDepth(p, -INFINITY, 100, 5, coldInfo, 0)
	if !PositionEqual(before, p) {
		t.Fatal("cold search state changed")
	}
	t.Logf("short=%d stores=%d rootTT=(%s,%d,depth%d,bound%d,hit%v) warm=%d warmNodes=%d cuts=%d cold=%d", short, shortInfo.QSearchTTStores, move.ToString(), score, depth, bound, hit, warm, warmInfo.Nodes, warmInfo.QTTCutoffs, cold)
	if warm >= 100 && cold < 100 {
		t.Fatal("inadequate capture budget produced contradicted bound")
	}
}

// A current-generation deep entry must survive a shallow colliding entry,
// including the first generation beyond the packed age field's maximum.
func TestReviewTTAgeWidth(t *testing.T) {
	for _, generations := range []int{0, 511, 512, 513, 1023, 1024} {
		cache := NewCache(1)
		for i := 0; i < generations; i++ {
			cache.AdvanceAge()
		}
		const deepHash uint64 = 1
		const shallowHash uint64 = 1 + (1 << 16)
		cache.Set(deepHash, EmptyMove, 100, 20, LowerBound, false)
		cache.Set(shallowHash, EmptyMove, 1, 1, LowerBound, false)
		_, _, _, _, hit, _ := cache.Get(deepHash)
		if !hit {
			t.Errorf("generation %d: shallow current-generation collision replaced depth20 entry", generations)
		}
	}
}
