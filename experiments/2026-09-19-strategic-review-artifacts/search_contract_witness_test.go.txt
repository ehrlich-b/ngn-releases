func reviewProbcutState(t *testing.T, depth int) (*Position, *SearchInfo, Move) {
	t.Helper()
	model := loadEvaluatorTensors(t, new(nnue.Tensors))
	searcher, err := NewSearchEngineWithHash(1)
	if err != nil {
		t.Fatal(err)
	}
	if err := searcher.SelectNGNV1Evaluator(model); err != nil {
		t.Fatal(err)
	}
	searcher.SetMaxNodes(1000)
	pos := n3cPosition(t, "4k3/p7/8/8/8/8/8/R3K3 w - - 0 1")
	searcher.sessionMu.Lock()
	generation := mustAcquireHCEModelUse()
	t.Cleanup(func() {
		releaseHCEModelUse()
		searcher.sessionMu.Unlock()
	})
	tt := searcher.prepareTTGeneration(generation)
	evaluator := searcher.mustPreparePrimaryEvaluator(pos, generation)
	info := newN3CDirectSearchInfo(searcher, evaluator, tt, depth)
	return pos, info, adapterLegalMove(t, pos, "a1a7")
}

func TestReviewSingularProbcutExcludedMoveWitness(t *testing.T) {
	pos, info, excluded := reviewProbcutState(t, 5)
	info.ExcludedMove, info.ExcludedPly = excluded, 0
	child := pos.Copy()
	child.MakeMove(excluded)
	childHash := child.Hash()
	info.tt.Set(childHash, EmptyMove, scoreToTT(0, 1), 1, Exact, false)
	visitedExcluded := false
	info.evaluator.transitionObserver = func(observed *Position, e *workerEvaluator) {
		if e.nnueDepth() == 1 && observed.Hash() == childHash {
			visitedExcluded = true
		}
	}
	score := alphaBetaPV(pos, 5, 0, -201, -200, false, false, false, info)
	if !visitedExcluded || info.ProbcutPrunes != 1 || score != -200 || info.Stopped {
		t.Fatalf("witness did not reproduce: excludedVisited=%v probcut=%d score=%d stopped=%v", visitedExcluded, info.ProbcutPrunes, score, info.Stopped)
	}
	if info.evaluator.nnueDepth() != 0 {
		t.Fatal("witness failed to unwind evaluator")
	}
	t.Logf("Excluded capture %s was searched by ProbCut at excluded ply 0 and certified cutoff %d; nodes=%d", excluded.ToString(), score, info.Nodes)
}

func TestReviewProbcutMoveStackWitness(t *testing.T) {
	pos, info, capture := reviewProbcutState(t, 6)
	stale := adapterLegalMove(t, pos, "e1d1")
	info.MoveStack[0] = stale
	observedGrandchild := false
	info.evaluator.transitionObserver = func(_ *Position, e *workerEvaluator) {
		// At frame 3 the ply-2 main-search node has already read prev2 and
		// selected a move. Its active root move was the ProbCut capture.
		if e.nnueDepth() == 3 && info.MoveStack[0] == stale {
			observedGrandchild = true
			t.Logf("ProbCut root capture %s reached a grandchild while MoveStack[0]=%s; frames=%d", capture.ToString(), info.MoveStack[0].ToString(), info.FrameDepth)
			info.control.RequestStop()
		}
	}
	score := alphaBetaPV(pos, 6, 0, -201, -200, false, false, false, info)
	if !observedGrandchild || !info.Stopped || score != stoppedSearchScore {
		t.Fatalf("witness did not reproduce: grandchild=%v stopped=%v score=%d nodes=%d", observedGrandchild, info.Stopped, score, info.Nodes)
	}
	if info.evaluator.nnueDepth() != 0 {
		t.Fatal("witness failed to unwind evaluator")
	}
}
